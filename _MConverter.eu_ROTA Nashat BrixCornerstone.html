const STORE_OPEN = 10;
const STORE_CLOSE = 18.5;

  Desire: { availability: { Wednesday:"10:00–15:00", Saturday:"10:00–16:00" } },

let staffColors = {

  Takudzwa:"#2980b9",

let rota = {

    Wednesday:{ hours:"10:00–18:30", staff:[{name:"Kwame",hours:"10:00–18:30"}] },
    Thursday: { hours:"10:00–18:30", staff:[{name:"Kwame",hours:"10:00–18:30"}] },
    Friday:   { hours:"10:00–18:30", staff:[] },
    Saturday: { hours:"10:00–18:30", staff:[
      {name:"Colin",hours:"10:00–16:00"},

  }
};
rota.week2 = JSON.parse(JSON.stringify(rota.week1));
rota.week3 = JSON.parse(JSON.stringify(rota.week1));
rota.week4 = JSON.parse(JSON.stringify(rota.week1));

let currentWeek = "week1";
let dragData = null;
let lockedDays = {};
let lockedStaff = {};
let preferredHours = {};

function setStyle(style) { document.body.className = style; }

function checkConflicts(day, staffName, hours) {
  const availability = staffProfiles[staffName]?.availability[day];
  if (!availability) return "full";
  const [aStart, aEnd] = availability.split("–");
  const [hStart, hEnd] = hours.split("–");
  const aS = parseFloat(aStart.replace(":", "."));
  const aE = parseFloat(aEnd.replace(":", "."));
  const hS = parseFloat(hStart.replace(":", "."));
  const hE = parseFloat(hEnd.replace(":", "."));
  if (hS >= aS && hE <= aE) return "none";
  if (hE <= aS || hS >= aE) return "full";
  return "partial";
}

function renderRota() {
  const week = rota[currentWeek];
  let html = `<table><thead><tr><th>Day</th><th>Hours</th><th>Staff</th></tr></thead><tbody>`;
  for (const day in week) {
    const entry = week[day];
    const staffButtons = entry.staff.map(s => {
      const name = s.name;
      const color = staffColors[name] || staffColors.Default;
      const conflict = entry.hours.includes("–") ? checkConflicts(day, name, s.hours) : "none";
      let conflictClass = "";
      if (conflict === "partial") conflictClass = "conflict-partial";
      if (conflict === "full") conflictClass = "conflict-full";
      return `
        <button class="staff-btn ${conflictClass}"
                style="background:${color};"
                draggable="true"
                ondragstart="onStaffDragStart(event,'${name}','${day}')"
                ondragend="onStaffDragEnd(event)"
                onclick="showStaffShift('${name}','${day}','${s.hours}')">
          ${name} (${s.hours})
        </button>
      `;
    }).join("");
    html += `<tr class="day-${day}">
      <td>${day}${lockedDays[day] ? " 🔒" : ""}</td>
      <td>${entry.hours}</td>
      <td ondragover="onStaffDragOver(event)" ondrop="onStaffDrop(event,'${day}')">
        ${staffButtons || "—"}
        <button class="edit-btn" onclick="openEditModal('${day}')">Edit</button>
      </td>
    </tr>`;
  }
  html += `</tbody></table>`;
  document.getElementById("rota-container").innerHTML = html;
}

function onStaffDragStart(e, name, day) {
  dragData = { name, day, fromPool:false };
  e.target.classList.add("dragging");
}
function onStaffDragEnd(e) { e.target.classList.remove("dragging"); }
function onStaffDragOver(e) { e.preventDefault(); }

function onStaffDrop(e, targetDay) {
  e.preventDefault();
  if (!dragData) return;
  const { name, day, fromPool } = dragData;
  const week = rota[currentWeek];

  if (!fromPool && day === targetDay) {
    const staffList = week[targetDay].staff;
    const draggedIndex = staffList.findIndex(s => s.name === name);
    const btn = e.target.closest(".staff-btn");
    if (btn) {
      const text = btn.textContent.trim();
      const afterName = text.split(" (")[0];
      const afterIndex = staffList.findIndex(s => s.name === afterName);
      if (draggedIndex !== -1 && afterIndex !== -1) {
        const [item] = staffList.splice(draggedIndex, 1);
        staffList.splice(afterIndex, 0, item);
      }
    }
    dragData = null;
    renderRota();
    return;
  }

  if (!fromPool) {
    week[day].staff = week[day].staff.filter(s => s.name !== name);
  }

  if (!week[targetDay].staff.find(s => s.name === name)) {
    const defaultHours = week[targetDay].hours.includes("–") ? week[targetDay].hours : "10:00–18:30";
    week[targetDay].staff.push({ name, hours: defaultHours });
  }

  dragData = null;
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
}

function addStaff() {
  const name = document.getElementById("new-staff-name").value.trim();
  if (!name) return;
  if (!staffProfiles[name]) staffProfiles[name] = { availability:{} };
  if (!staffColors[name]) staffColors[name] = staffColors.Default;
  document.getElementById("new-staff-name").value = "";
  populateStaffDropdown();
  renderStaffPool();
}

function populateStaffDropdown() {
  const select = document.getElementById("staff-select");
  select.innerHTML = "";
  Object.keys(staffProfiles).forEach(name => {
    const opt = document.createElement("option");
    opt.value = name;
    opt.textContent = name + (lockedStaff[name] ? " 🔒" : "");
    select.appendChild(opt);
  });
}

function renderStaffPool() {
  const pool = document.getElementById("staff-pool");
  pool.innerHTML = "";
  Object.keys(staffProfiles).forEach(name => {
    const div = document.createElement("div");
    div.className = "staff-pool-item";
    div.textContent = name + (lockedStaff[name] ? " 🔒" : "");
    div.draggable = true;
    div.ondragstart = (e) => { dragData = { name, fromPool:true }; };
    pool.appendChild(div);
  });
}

function addExtraStaff() {

function renderStaffTimeline() {
  const name = document.getElementById("staff-select").value;
  if (!name) { document.getElementById("timeline-view").innerHTML=""; return; }
  const week = rota[currentWeek];
  const startHour = STORE_OPEN;
  const endHour = STORE_CLOSE;
  const hourWidth = 20;
  let html = `<p><strong>${name}</strong>${lockedStaff[name] ? " 🔒" : ""}</p>`;
  for (const day in week) {
    const entry = week[day];
    const staffEntry = entry.staff.find(s => s.name === name);
    html += `<div>${day}: `;
    if (staffEntry) {
      html += `<span style="display:inline-block;background:${staffColors[name]||staffColors.Default};height:10px;width:${(endHour-startHour)*hourWidth}px;"></span> ${staffEntry.hours}`;
    } else {
      html += `Not working`;
    }
    html += `</div>`;
  }
  if (preferredHours[name]) {
    html += `<div>Preferred: ${preferredHours[name].avgStart}–${preferredHours[name].avgEnd}</div>`;
  }
  document.getElementById("timeline-view").innerHTML = html;
}

function autoGenerateRota() {
  const week = rota[currentWeek];
  for (const day in week) {
    if (lockedDays[day]) continue;
    week[day].staff = [];
    for (const name in staffProfiles) {
      if (lockedStaff[name]) continue;
      const profile = staffProfiles[name];
      if (profile.availability[day]) {
        week[day].staff.push({ name, hours: week[day].hours });
      }
    }
    if (!week[day].hours.includes("–") && day !== "Sunday") {
      week[day].hours = "10:00–18:30";
    }
  }
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
}

function rotateStaffDays() {
  const week = rota[currentWeek];
  const days = Object.keys(week);
  let prev = week[days[days.length-1]].staff;
  days.forEach(day => {
    if (lockedDays[day]) return;
    const temp = week[day].staff;
    week[day].staff = prev;
    prev = temp;
  });
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
}

function build24HourDropdown(id) {
  const select = document.getElementById(id);
  select.innerHTML = "";
  for (let h=0; h<24; h++) {
    for (let m of ["00","30"]) {
      const time = `${String(h).padStart(2,"0")}:${m}`;
      const opt = document.createElement("option");
      opt.value = time;
      opt.textContent = time;
      select.appendChild(opt);
    }
  }
}

  document.getElementById(`tab-content-${tab}`).classList.add('active');
}

function saveIndividualHours() {
  const day = document.getElementById("modal-day").value;
  const entry = rota[currentWeek][day];
  entry.staff.forEach(s => {
    const st = document.getElementById(`staff-start-${s.name}`).value;
    const en = document.getElementById(`staff-end-${s.name}`).value;
    s.hours = `${st}–${en}`;
    if (!preferredHours[s.name]) {
      preferredHours[s.name] = {
        count: 0,
        totalStart: 0,
        totalEnd: 0,
        avgStart: st,
        avgEnd: en
      };
    }
    const ph = preferredHours[s.name];
    ph.count++;
    const stNum = parseFloat(st.replace(":", "."));
    const enNum = parseFloat(en.replace(":", "."));
    ph.totalStart += stNum;
    ph.totalEnd += enNum;
    ph.avgStart = (ph.totalStart / ph.count).toFixed(2).replace(".", ":");
    ph.avgEnd = (ph.totalEnd / ph.count).toFixed(2).replace(".", ":");
  });
  alert("Individual hours saved and preferences updated");
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
}

function applyModalEdit() {
  const day = document.getElementById("modal-day").value;
  if (lockedDays[day]) { alert("Day is locked"); return; }
  const start = document.getElementById("modal-start").value;
  const end = document.getElementById("modal-end").value;
  rota[currentWeek][day].hours = `${start}–${end}`;
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
}

function addStaffToDay() {
  const day = document.getElementById("modal-day").value;
  if (lockedDays[day]) { alert("Day is locked"); return; }
  const selected = Array.from(document.getElementById("modal-staff").selectedOptions).map(o => o.value);
  const entry = rota[currentWeek][day];
  selected.forEach(name => {
    if (!lockedStaff[name] && !entry.staff.find(s => s.name === name)) {
      entry.staff.push({ name, hours: entry.hours });
    }
  });
  renderIndividualHours(day);
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
  populateAvailabilitySelect(day);
}

function removeSelectedStaff() {
  const day = document.getElementById("modal-day").value;
  if (lockedDays[day]) { alert("Day is locked"); return; }
  const selected = Array.from(document.getElementById("modal-staff").selectedOptions).map(o => o.value);
  const entry = rota[currentWeek][day];
  entry.staff = entry.staff.filter(s => !selected.includes(s.name) || lockedStaff[s.name]);
  renderIndividualHours(day);
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
  populateAvailabilitySelect(day);
}

function closeModal() { document.getElementById("edit-modal").style.display = "none"; }

function copyDay() {
  const day = prompt("Enter day to copy FROM (e.g. Monday)");
  const toDay = prompt("Enter day to copy TO (e.g. Tuesday)");
  if (!rota[currentWeek][day] || !rota[currentWeek][toDay]) { alert("Invalid day name"); return; }
  if (lockedDays[toDay]) { alert("Target day is locked"); return; }
  rota[currentWeek][toDay] = JSON.parse(JSON.stringify(rota[currentWeek][day]));
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
}

function copyWeek() {
  const from = prompt("Copy FROM week number (1–4)");
  const to = prompt("Copy TO week number (1–4)");
  if (!rota[`week${from}`] || !rota[`week${to}`]) { alert("Invalid week number"); return; }
  rota[`week${to}`] = JSON.parse(JSON.stringify(rota[`week${from}`]));
  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();

function suggestShifts() {
  const week = rota[currentWeek];
  let totals = {};
  for (const day in week) {
    const entry = week[day];
    entry.staff.forEach(s => {
      const [st, en] = s.hours.split("–");
      const stH = parseFloat(st.replace(":", "."));
      const enH = parseFloat(en.replace(":", "."));
      const hours = enH - stH;
      totals[s.name] = (totals[s.name] || 0) + hours;
    });
  }

  for (const day in week) {
    if (lockedDays[day]) continue;
    const entry = week[day];
    if (!entry.hours.includes("–")) continue;
    const [hStart, hEnd] = entry.hours.split("–");
    const hS = parseFloat(hStart.replace(":", "."));
    const hE = parseFloat(hEnd.replace(":", "."));
    const shiftHours = hE - hS;

    let candidates = [];
    for (const name in staffProfiles) {
      if (lockedStaff[name]) continue;
      const avail = staffProfiles[name].availability[day];
      if (!avail) continue;
      const [aStart, aEnd] = avail.split("–");
      const aS = parseFloat(aStart.replace(":", "."));
      const aE = parseFloat(aEnd.replace(":", "."));
      if (hS >= aS && hE <= aE) {
        let preferenceScore = 0;
        if (preferredHours[name]) {
          const ph = preferredHours[name];
          const pS = parseFloat(ph.avgStart.replace(":", "."));
          const pE = parseFloat(ph.avgEnd.replace(":", "."));
          const startDiff = Math.abs(pS - hS);
          const endDiff = Math.abs(pE - hE);
          preferenceScore = startDiff + endDiff;
        }
        candidates.push({ name, currentHours: totals[name] || 0, preferenceScore });
      }
    }

    candidates.sort((a, b) => {
      if (a.preferenceScore !== b.preferenceScore) return a.preferenceScore - b.preferenceScore;
      return a.currentHours - b.currentHours;
    });

    const chosen = candidates.slice(0, 2).map(c => c.name);
    entry.staff = chosen.map(name => ({ name, hours: entry.hours }));
    chosen.forEach(name => {
      totals[name] = (totals[name] || 0) + shiftHours;
    });
  }

  renderRota();
  renderStaffTimeline();
  renderSummary();
  renderCalendar();
  renderPrintWeeks();
  alert("AI suggestions applied using preferred hours and availability (locks respected)");
}

  for (let w = 1; w <= 4; w++) {

    const week = rota[weekKey];
    let html = `<h4>Week ${w}</h4>`;
    html += `<table><thead><tr><th>Day</th><th>Hours</th><th>Staff</th></tr></thead><tbody>`;
    for (const day in week) {

        </tr>
      `;
    }
    html += `</tbody></table><br>`;
    container.innerHTML += html;
  }

      rota = data.rota;
      staffProfiles = data.staffProfiles;
      staffColors = data.staffColors;
      lockedDays = data.lockedDays;
      lockedStaff = data.lockedStaff;

      renderStaffTimeline();
      renderSummary();
      renderCalendar();
      renderPrintWeeks();

    } catch (err) {
      alert("Invalid config file");
    }
  };

  reader.readAsText(file);
}
